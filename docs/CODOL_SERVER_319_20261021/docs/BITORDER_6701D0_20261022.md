## 0. 结论先行：位序是什么

`0x6701D0` 本身**只负责一个字段的"值"部分**，不含索引。
"变位标志 + 索引 + 值"这三段分布在三个函数里：

| 层 | 函数 | 负责 |
|---|---|---|
| 结构层 | `0x670FF0` | has-delta 位 + 最后变化索引 + 游程 |
| 索引层 | `0x670CB0` | 读"跳过多少个未变字段" |
| 字段层 | `0x6701D0` | 读单个字段：1 个变位标志 + 该 codec 的负载 |

## 1. 0x6701D0 的调用约定（已逐条验证）

```
ecx (thiscall this) = netField_t* entry   ← 不是表基址，是 entry 指针
[ebp+8]  = msg*        (位流)
[ebp+9]  = to struct*  (目标)
[ebp+10] = from struct*(基线)
[ebp+14] = absolute 标志 (byte)  1 = 忽略 from，用全零基线
[ebp+18] = time        (供时间类 codec 用)
```

入口处三条指令即确定：

```asm
006701E0  cmp byte ptr [ebp+0x14],0     ; absolute?
006701E4  mov eax,[ebp+0x10]            ; from
006701ED  mov [ebp-0x14],ecx            ; save entry
006701F7  je  0x6701FE
006701F9  lea edi,[ebp-0x20]            ; edi = &zero   (absolute)
006701FC  jmp 0x670202
006701FE  mov edi,[ecx]                 ; edi = entry->offset
00670200  add edi,eax                   ; edi = from + offset
00670202  mov ebx,[ecx]                 ; ebx = entry->offset
00670204  mov ecx,[ecx+4]               ; ecx = entry->bits
00670207  add ebx,edx                   ; ebx = to + offset
```

即：`edi` = 基线字段指针（或全零），`ebx` = 目标字段指针，`ecx` = codec。
**后续所有分支只做一件事：把读到的位流写进 `[ebx]`。**

## 2. 结构层位序（0x670FF0，协议骨架）

```asm
00670FFC  call 0x447190        ; bit0 = hasDelta
00671005  je   0x671140        ; hasDelta==0 -> 走全量
0067100E  bsr  eax,[ebp+0x18]  ; numFields
00671019  xor  eax,0x1f
0067101C  mov  ecx,0x20
00671021  sub  ecx,eax         ; indexBits = floor(log2(n))+1
00671026  call 0x66EC30        ; read indexBits -> lastChangedIndex
```

```
┌─────────────────────────────────────────────────────────────┐
│ bit 0            : hasDelta                                 │
│  hasDelta == 0   : 全量结构，逐字段读，无任何变位标志        │
│  hasDelta == 1   :                                          │
│    indexBits     : lastChangedIndex                         │
│    循环 {                                                   │
│        varint    : run  (连续未变字段个数)                   │
│        run 个字段: 直接读值，**不带变位标志**                │
│        1 个字段  : 读该字段的变位标志 + 值                   │
│    } 直到 index == lastChangedIndex                          │
│    剩余字段       : 逐个读变位标志 + 值                      │
└─────────────────────────────────────────────────────────────┘
```

`indexBits`：87→7，24→5，9→4，207→8（`floor(log2(n))+1`）。

## 3. 索引层 varint（0x670CB0）

```asm
00670CBB  cmp ebx,1            ; indexBits == 1 ?
00670CC0  ... 循环 inc edi / call 0x447190 / je  ; 一元码：数连续的 1
00670CD4  ... 否则：
00670CD9  call 0x447190        ; 1 bit
00670CE2  -> return 1          ; 该位为 1 => run = 1
00670CEC  push ebx
00670CF4  call 0x66EC30        ; 读 indexBits 位
00670CFD  shl esi,cl / dec esi ; mask = (1<<indexBits)-1
00670D03  cmp eax,esi
00670D07  循环：v==mask 时继续累加，v!=mask 时结束
```

**语义**：`run = 1 + Σ(v_i)`，遇到 `v_i != mask` 停止。
即"再跳过多少个字段才到下一个变化字段"。

## 4. 字段层：45 个 codec 的负载位宽（0x6701D0 全扫）

### 4.1 `bits == 0` —— float（0x67020D）

```
1 bit  : 0
  1 bit : b
     value = b << 31          ; ±0.0
1 bit  : 1
  1 bit  : 1 -> 32 bits raw, XOR from
  1 bit  : 0 -> 5 bits mantissa + 5 bits exponent（压缩浮点）
```

### 4.2 `bits > 0` —— 默认整数路径（0x670AF2 → 0x670B15）

```
1 bit  : changed
  == 0 : value = 0
  == 1 : ReadBits(bits) XOR from      (0x66FE30 做符号扩展)
```

## 5. 与本项目 snapshot.c 的对照

当前 `src/battle/snapshot.c` 的写法：

```c
static void bw_index(bw_t *w, int index, int *prev)
{
    int skip = index - *prev - 1;
    *prev = index;
    if (skip == 0) { bw_bits(w, 1, 1); return; }
    bw_bits(w, 0, 1);
    while (skip >= 15) { bw_bits(w, 15, 4); skip -= 15; }
    bw_bits(w, (uint32_t)skip, 4);
}
```

**历史实现错误**：这是把 `indexBits` 硬编码成 4 位、把 varint 写成 4 位一组。2026-10-23 已修正为按表宽编码，并新增 `tests/test_snapshot_run.c`。
3.19 的 playerState 是 **7 位**，`0x670CB0` 读的是 `indexBits` 位一组，
且**首个 1 bit 为 1 时 run=1**（不是 skip==0 的语义）。

正确实现：

```c
/* indexBits = floor(log2(numFields)) + 1 */
static void bw_run(bw_t *w, int run, int indexBits)
{
    if (indexBits == 1) {              /* 一元码 */
        for (int i = 0; i < run; i++) bw_bits(w, 1, 1);
        bw_bits(w, 0, 1);
        return;
    }
    if (run == 1) { bw_bits(w, 1, 1); return; }
    bw_bits(w, 0, 1);
    int left = run - 1;
    int mask = (1 << indexBits) - 1;
    while (left >= mask) { bw_bits(w, mask, indexBits); left -= mask; }
    bw_bits(w, left, indexBits);
}
```
